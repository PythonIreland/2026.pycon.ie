{{- /* JSON-LD for /speakers/: an ItemList of Person nodes built from Sessionize. Each Person has a stable @id (speakers/#) referenced by the sessions on /schedule/ and by the main Event's performer list, and links back to its sessions via performerIn. */ -}} {{- $sz := partialCached "sessionize-data.html" . -}} {{- if $sz.ok }} {{- $base := site.BaseURL -}} {{- $items := slice -}} {{- range $i, $s := $sz.speakers -}} {{- $person := dict "@type" "Person" "@id" (printf "%sspeakers/#%s" $base $s.id) "name" $s.name -}} {{- with $s.tagline }}{{ $person = merge $person (dict "jobTitle" .) }}{{ end -}} {{- with $s.bio }}{{ $person = merge $person (dict "description" .) }}{{ end -}} {{- with $s.photo }}{{ $person = merge $person (dict "image" .) }}{{ end -}} {{- with $s.links }}{{ $person = merge $person (dict "sameAs" .) }}{{ end -}} {{- $performerIn := slice -}} {{- range $s.sessions }}{{ $performerIn = $performerIn | append (dict "@id" (printf "%sschedule/#session-%s" $base .)) }}{{ end -}} {{- with $performerIn }}{{ $person = merge $person (dict "performerIn" .) }}{{ end -}} {{- $items = $items | append (dict "@type" "ListItem" "position" (add $i 1) "item" $person) -}} {{- end }} {{- end }}
